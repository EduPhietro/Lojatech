6. Explique a diferença entre:
	a) CHAR e VARCHAR: CHAR aloca o espaço definido pelo limite * o tamanho do caractere que é 1byte, já o VARCHAR aloca dinamicamente o tamanho, mas limita o tamanho ao limite;
	b) INT e DECIMAL/NUMERIC: INT representa o conjunto dos números inteiros, o DECIMAL/NUMERIC são supostamente a mesma coisa, representam o conjunto dos Decimais; DECIMAL é alias de NUMERIC;
	c) DATE, TIME e TIMESTAMP: DATE armazena apenas datas no formato ISO 8601; TIME armazena horas e frações de horas no formato ISO 8601, TIMESTAMP armazena DATA e HORA no formato ISO 8601;

	d) NULL e vazio (""): NULL representa ausencia total de valor ou referência enquanto uma String vazia é um dado armazenado válido;

7. Justifique por que não é recomendado utilizar o tipo FLOAT para armazenar valores monetários
em sistemas comerciais.
	R: FLOAT é um tipo de dado utilizado para representar dados no formato decimal, regulamentado pela IEEE 754, entretanto ele tem um tamanho limitado para representar dados decimais, 
	o FLOAT pode representar dados de até 32 bits, sendo 1 bit para representar a sua posição na reta numérica, negativo ou positivo, 8 bits para representar um exponente e 23 bits
	para representar  a fração; todos os números são representados na base 2; Não é possível representar valores monetários porque dízimas periódicas não cabem, ou valores decimais muito pequenos não podem ser representados. como o 0.3.

	-- 3. CRIAÇÃO DO BANCO DE DADOS E TABELAS
	CREATE DATABASE lojatech;

	-- Tabela de Clientes
	CREATE TABLE IF NOT EXISTS clientes ( 
		id_cliente INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
		nome VARCHAR(100) NOT NULL,
		email VARCHAR(150) NOT NULL UNIQUE,
		telefone VARCHAR(20) NULL,
		data_nascimento DATE NULL,
		cidade VARCHAR(80) NULL,
		data_cadastro TIMESTAMP NOT NULL DEFAULT now()
	);

	-- Tabela de Produtos
	CREATE TABLE IF NOT EXISTS produtos (
		id_produto INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
		nome_produto VARCHAR(100) NOT NULL,
		categoria VARCHAR(50) NOT NULL,
		preco_unitario DECIMAL(10,2) NOT NULL CHECK (preco_unitario > 0),
		estoque INT NOT NULL DEFAULT 0 CHECK (estoque >= 0),
		descricao TEXT NULL,
		ativo BOOLEAN NOT NULL DEFAULT TRUE
	);

	-- Tabela de Pedidos
	CREATE TABLE IF NOT EXISTS pedidos (
		id_pedido INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
		id_cliente INT NOT NULL,
		data_pedido TIMESTAMP NOT NULL DEFAULT now(),
		status VARCHAR(20) NOT NULL DEFAULT 'ABERTO' CHECK (status IN ('PAGO','ABERTO','CANCELADO')),
		valor_total DECIMAL(10,2) NOT NULL DEFAULT 0 CHECK (valor_total >=0),
		
		CONSTRAINT pedidos_clientes_fk 
		FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente)
			ON DELETE CASCADE
	);

	-- Tabela de Itens do Pedido
	CREATE TABLE IF NOT EXISTS itens_pedido (
		id_item INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
		id_pedido INT NOT NULL,
		id_produto INT NOT NULL,
		quantidade INT NOT NULL CHECK (quantidade > 0),
		preco_unitario DECIMAL(10,2) NOT NULL CHECK (preco_unitario > 0),

		CONSTRAINT item_pedido_fk 
		FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido) 
			ON DELETE CASCADE,
		
		CONSTRAINT item_produto_fk 
		FOREIGN KEY (id_produto) REFERENCES produtos(id_produto)
			ON DELETE CASCADE
	);

	Consulta:
		WITH valor_total AS (
			SELECT
			ip.id_pedido,
			SUM((p.preco_unitario * ip.quantidade)) AS valor_total
			FROM itens_pedido ip 
				LEFT JOIN produtos p ON ip.id_produto  = p.id_produto 
			GROUP  BY id_pedido
		)
			SELECT 
				c.nome,
				p.id_pedido,
				COALESCE(tv.valor_total,0) AS valor_total  
			FROM pedidos p

				LEFT JOIN clientes c  ON p.id_cliente  = c.id_cliente;
